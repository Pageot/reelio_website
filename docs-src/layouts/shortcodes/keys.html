{{- /* Shortcut table. One line per row: `KEYS | description`. In KEYS, tokens are separated by spaces; a token joined by "+" becomes one per key ("Ctrl+Shift+Tab"). The words "or", "to", "then" and "/" stay plain text. An underscore inside a key shows as a space ("Side_button"). The description is inline Markdown. */ -}} {{- $connectors := slice "or" "to" "then" "/" "," -}}
{{- range split (trim .Inner "\n ") "\n" }} {{- $line := trim . " " }} {{- if $line }} {{- $parts := split $line "|" }} {{- if ne (len $parts) 2 }}{{ errorf "keys shortcode: expected `KEYS | description`, got %q" $line }}{{ end }} {{- $keys := trim (index $parts 0) " " }} {{- $desc := trim (index $parts 1) " " }}
{{- range $t := split $keys " " }} {{- if $t }} {{- if in $connectors $t }}{{ $t }} {{- else }} {{- range $k, $key := split $t "+" }}{{ if $k }}+{{ end }}{{ replace $key "_" " " }}{{ end }} {{- end }} {{- end }} {{- end }}
{{ $.Page.RenderString $desc }}
{{- end }} {{- end }}
